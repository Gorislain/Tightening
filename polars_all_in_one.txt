# Polars: ТОП функций по темам

**146 полезных функций и методов**, сгруппированных по задачам.

Документация: https://docs.pola.rs/api/python/stable/reference/

> Это справочник примеров: `df`, `lf`, `other` и т. д. — переменные вашего проекта. Некоторые функции зависят от версии Polars.

## 01. Чтение и запись

| Функция | Описание | Пример |
|---|---|---|
| `pl.read_csv` | Читает CSV-файл сразу в DataFrame. | `pl.read_csv("data.csv")` |
| `pl.scan_csv` | Лениво читает CSV: данные загружаются при collect(). | `pl.scan_csv("data.csv").collect()` |
| `pl.read_parquet` | Читает файл Parquet. | `pl.read_parquet("data.parquet")` |
| `pl.scan_parquet` | Лениво читает Parquet с оптимизацией запроса. | `pl.scan_parquet("data.parquet").collect()` |
| `pl.read_json` | Читает JSON-документ. | `pl.read_json("data.json")` |
| `pl.read_ndjson` | Читает JSON Lines, одна запись в строке. | `pl.read_ndjson("data.ndjson")` |
| `pl.scan_ndjson` | Лениво читает построчный JSON. | `pl.scan_ndjson("data.ndjson").collect()` |
| `pl.read_excel` | Читает данные из Excel, может потребовать доп. пакет. | `pl.read_excel("data.xlsx")` |
| `pl.read_database` | Выполняет SQL и читает таблицу из соединения БД. | `pl.read_database("SELECT * FROM t", connection)` |
| `pl.read_database_uri` | Читает данные БД по URI с подходящим движком. | `pl.read_database_uri("SELECT * FROM t", uri)` |
| `pl.DataFrame.write_csv` | Сохраняет DataFrame в CSV. | `df.write_csv("out.csv")` |
| `pl.DataFrame.write_parquet` | Сохраняет DataFrame в Parquet. | `df.write_parquet("out.parquet")` |
| `pl.DataFrame.write_json` | Сохраняет DataFrame в JSON. | `df.write_json("out.json")` |
| `pl.DataFrame.write_ndjson` | Сохраняет DataFrame в JSON Lines. | `df.write_ndjson("out.ndjson")` |
| `pl.DataFrame.write_excel` | Экспортирует данные в Excel (может потребовать xlsxwriter). | `df.write_excel("out.xlsx")` |

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Генератор справочника функций Polars на русском языке.

Назначение
----------
1. Создать ТОП полезных функций Polars с пояснениями и короткими примерами.
2. Найти ВСЕ доступные публичные функции/методы установленной версии Polars:
   pl.*, DataFrame, LazyFrame, Series, Expr, их пространства имён,
   GroupBy, LazyGroupBy, selectors и SQLContext (если доступны).
3. Сохранить результаты в Markdown и CSV, чтобы читать их на GitHub и ПК.

Установка: python -m pip install -U polars
Примеры:
    python polars_reference_generator.py
    python polars_reference_generator.py --output docs/polars_api.md
    python polars_reference_generator.py --top 50
    python polars_reference_generator.py --search join
    python polars_reference_generator.py --only-top
    python polars_reference_generator.py --demo

Примечание: описания ТОП-функций написаны на русском. Для остальных
функций используется первая строка официального docstring установленного

def plain_doc(obj: Any) -> str:
    """Первая информативная строка docstring с ограничением длины."""
    try:
        source = inspect.getdoc(obj) or ""
    except (ValueError, TypeError, AttributeError):
        source = ""
    for line in source.splitlines():
        candidate = line.strip()
        if not candidate or candidate.startswith(("..", ">>>", "Parameters", "Notes")):
            continue
        candidate = re.sub(r"\s+", " ", candidate)
        return candidate[:260].rstrip()
    return "Документация не предоставлена в установленной версии Polars."


def members(obj: Any) -> Iterator[tuple[str, Any]]:
    """Публичные вызываемые атрибуты без выполнения самих функций."""
    try:
        names = dir(obj)
    except Exception:
        return
    for name in names:
        if name.startswith("_"):
            continue
        try:
            item = getattr(obj, name)
        except Exception:
            continue
        if callable(item):
            yield name, item


def available_targets() -> Iterator[tuple[str, str, Any]]:
    """Объекты API, которые можно перечислить без чтения документации из сети."""
    yield ("Функции верхнего уровня", "pl", pl)
    for cls_name in ("DataFrame", "LazyFrame", "Series", "Expr"):
        cls = getattr(pl, cls_name, None)
        if cls is not None:
            yield (f"Методы {cls_name}", f"pl.{cls_name}", cls)

    # Пространства имён .str, .dt, .list, .arr, .struct и др.
    try:
        expr = pl.col("_column")
    except Exception:
        expr = None
    series = None
    try:
        series = pl.Series("_column", [1])
    except Exception:
        pass
    for type_name, base in (("Expr", expr), ("Series", series)):
        if base is None:
            continue
        for namespace in ("str", "dt", "list", "arr", "struct", "bin", "cat", "name", "meta"):
            try:
                accessor = getattr(base, namespace)
            except Exception:
                continue
            yield (
                f"{type_name}: пространство имён .{namespace}",
                f"pl.{type_name}.{namespace}",
                accessor,
            )

    try:
        import polars.selectors as cs
        yield ("Селекторы колонок", "pl.selectors", cs)
    except ImportError:
        pass

    try:
        group_by = pl.DataFrame({"_key": [1], "_v": [2]}).group_by("_key")
        yield ("Методы GroupBy", "pl.GroupBy", group_by)
    except Exception:
        pass
    try:
        lazy_group_by = pl.DataFrame({"_key": [1]}).lazy().group_by("_key")
        yield ("Методы LazyGroupBy", "pl.LazyGroupBy", lazy_group_by)
    except Exception:
        pass
    try:
        sql_context = pl.SQLContext()
        yield ("Методы SQLContext", "pl.SQLContext", sql_context)
    except Exception:
        pass


def create_catalog() -> dict[str, list[tuple[str, Any]]]:
    groups: dict[str, list[tuple[str, Any]]] = {}
    visited: set[str] = set()
    for group, prefix, target in available_targets():
        items: list[tuple[str, Any]] = []
        for name, item in members(target):
            full_name = f"{prefix}.{name}"
            if full_name in visited:
                continue
            if prefix == "pl" and isinstance(item, type):
                # Классы отдельно представлены в ТОПе, здесь — только функции.
                continue
            visited.add(full_name)
            items.append((full_name, item))
        if items:
            groups[group] = sorted(items, key=lambda pair: pair[0].casefold())
    return groups


def escape_table(value: str) -> str:
    return value.replace("|", r"\|").replace("\n", " ").replace("`", "'")


def filter_text(term: str, *parts: str) -> bool:
    return not term or term.casefold() in " ".join(parts).casefold()


def produce(top_limit: int, search: str, include_full: bool) -> tuple[str, list[tuple[str, str, str, str]]]:
    catalogue = create_catalog() if include_full else {}
    existing = {name for names in catalogue.values() for name, _ in names}
    by_name = {name: item for names in catalogue.values() for name, item in names}
    top_lookup = {name: (desc, example) for rows in TOPICS.values() for name, desc, example in rows}
    date = datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC")
    markdown = [
        "# Polars: ТОП-функции и полный справочник Python API",
        "",
        f"**Установленная версия Polars:** `{pl.__version__}`  ",
        f"**Сгенерировано:** {date}  ",
        "**Документация:** https://docs.pola.rs/api/python/stable/reference/",
        "",
        "Функции сгруппированы по задачам. ТОП снабжён русскими пояснениями и мини-примерами. ",
        "Полный каталог определяется автоматически по установленному Polars и содержит ",
        "описания из docstring библиотеки, если русского пояснения нет.",
        "",
        "> Примеры — краткие шаблоны: переменные `df`, `other`, `lf`, `start`, `end` и `connection` ",
        "> требуется создать самостоятельно. Методы, отсутствующие в вашей версии, ",
        "> отмечены как недоступные только при генерации полного каталога.",
        "",
    ]
    rows_for_csv: list[tuple[str, str, str, str]] = []
    count_top = 0
    markdown += ["## Часть I. ТОП самых полезных функций по темам", ""]
    for topic, records in TOPICS.items():
        filtered = []
        for name, ru, example in records:
            if not filter_text(search, name, ru, example, topic):
                continue
            if count_top >= top_limit:
                break
            filtered.append((name, ru, example))
            count_top += 1
        if not filtered:
            continue
        markdown += [f"### {topic}", "", "| Функция | Описание | Пример |", "|---|---|---|"]
        for name, ru, example in filtered:
            suffix = ""
            if include_full and name not in existing and name not in ("pl.DataFrame", "pl.Series", "pl.LazyFrame", "pl.DataFrame.schema"):
                suffix = " ⚠️ *нет в версии*"
            markdown.append(
                f"| `{escape_table(name)}`{suffix} | {escape_table(ru)} | `{escape_table(example)}` |"
            )
            rows_for_csv.append(("ТОП / " + topic, name, ru, example))
        markdown.append("")
    markdown += [f"Всего в ТОПе: **{count_top}** функций и свойств.", ""]

    if include_full:
        markdown += ["## Часть II. Все обнаруженные публичные функции и методы", ""]
        all_count = 0
        for group, entries in catalogue.items():
            filtered = []
            for name, obj in entries:
                desc = top_lookup.get(name, (None, None))[0] or plain_doc(obj)
                if filter_text(search, name, desc, group):
                    filtered.append((name, obj, desc))
            if not filtered:
                continue
            markdown += [f"### {group}", "", "| Функция / метод | Описание |", "|---|---|"]
            for name, _, desc in filtered:
                markdown.append(f"| `{escape_table(name)}` | {escape_table(desc)} |")
                rows_for_csv.append((group, name, desc, ""))
                all_count += 1
            markdown.append("")
        markdown += [f"**Всего найдено методов и функций:** {all_count}.", ""]
        markdown += [
            "Для детальной сигнатуры, параметров и примеров используйте официальную ",
            "документацию или в Python: `help(pl.Expr.rolling_mean)`.",
            "",
        ]
    return "\n".join(markdown), rows_for_csv


def run_demo() -> None:
    """Небольшой выполняемый пример с популярными выражениями Polars."""
    df = pl.DataFrame(
        {
            "category": ["GPU", "CPU", "GPU", "SSD", "CPU"],
            "price": [200_000, 150_000, 260_000, 40_000, None],
            "qty": [2, 1, 1, 3, 5],
        }
    )
    print("\n=== Оригинальная таблица ===")
    print(df)
    print("\n=== ТОП-операции: filter, with_columns, group_by, agg, sort ===")
    print(
        df.lazy()
        .with_columns(pl.col("price").fill_null(0).alias("price"))
        .filter(pl.col("qty") >= 1)
        .with_columns((pl.col("price") * pl.col("qty")).alias("revenue"))
        .group_by("category")
        .agg(
            pl.len().alias("rows"),
            pl.col("revenue").sum().alias("revenue_sum"),
            pl.col("price").mean().round(2).alias("avg_price"),
        )
        .sort("revenue_sum", descending=True)
        .collect()
    )


def main() -> None:
    parser = argparse.ArgumentParser(description="ТОП и полный справочник функций Polars")
    parser.add_argument("--output", default="polars_functions.md", help="Куда записать Markdown")
    parser.add_argument("--top", type=int, default=250, help="Максимум функций в ТОПе")
    parser.add_argument("--search", default="", help="Поиск имени/описания")
    parser.add_argument("--only-top", action="store_true", help="Не собирать полный API")
    parser.add_argument("--demo", action="store_true", help="Выполнить учебный пример")
    args = parser.parse_args()

    if args.top < 0:
        parser.error("--top должен быть неотрицательным")
    if args.demo:
        run_demo()

    doc, csv_rows = produce(args.top, args.search.strip(), not args.only_top)
    md_path = Path(args.output).expanduser()
    md_path.parent.mkdir(parents=True, exist_ok=True)
    md_path.write_text(doc, encoding="utf-8")
    csv_path = md_path.with_suffix(".csv")
    with csv_path.open("w", encoding="utf-8-sig", newline="") as out:
        writer = csv.writer(out, delimiter=";")
        writer.writerow(["Раздел", "Функция", "Описание", "Пример"])
        writer.writerows(csv_rows)

    print(f"Версия Polars: {pl.__version__}")
    print(f"Записано: {md_path.resolve()}")
    print(f"Записано: {csv_path.resolve()}")
    print(f"Строк каталога и ТОПа: {len(csv_rows)}")
    if args.search:
        print(f"Поисковый запрос: {args.search!r}")


if __name__ == "__main__":
    main()
